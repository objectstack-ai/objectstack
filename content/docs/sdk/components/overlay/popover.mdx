## Examples

**Basic Popover**

```json
{
  "type": "popover",
  "trigger": {
    "type": "button",
    "label": "Open Popover"
  },
  "content": [
    {
      "type": "text",
      "content": "Popover content"
    }
  ]
}
```

## Schema

```ts
import type { SchemaNode } from '@object-ui/types';

interface PopoverSchema {
  type: 'popover';
  trigger: SchemaNode | SchemaNode[];
  content?: SchemaNode | SchemaNode[];
  side?: 'top' | 'right' | 'bottom' | 'left';
}
```
