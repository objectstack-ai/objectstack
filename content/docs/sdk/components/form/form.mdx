The Form component provides a complete form solution with validation and submission handling.

## Basic Usage

**Login Form**

```json
{
  "type": "form",
  "fields": [
    {
      "name": "email",
      "label": "Email",
      "type": "input",
      "inputType": "email",
      "placeholder": "Enter your email",
      "required": true
    },
    {
      "name": "password",
      "label": "Password",
      "type": "input",
      "inputType": "password",
      "placeholder": "Enter your password",
      "required": true
    }
  ],
  "submitLabel": "Sign In"
}
```

## With Validation

**Registration Form**

```json
{
  "type": "form",
  "fields": [
    {
      "name": "username",
      "label": "Username",
      "type": "input",
      "placeholder": "Choose a username",
      "required": true,
      "minLength": 3,
      "maxLength": 20
    },
    {
      "name": "email",
      "label": "Email",
      "type": "input",
      "inputType": "email",
      "required": true,
      "pattern": "^[^@]+@[^@]+\\.[^@]+$"
    },
    {
      "name": "age",
      "label": "Age",
      "type": "input",
      "inputType": "number",
      "min": 18,
      "max": 120
    }
  ],
  "submitLabel": "Register"
}
```

## Schema

```ts
interface FormField {
  name: string;
  label?: string;
  type: 'input' | 'textarea' | 'select' | 'checkbox' | 'radio-group' | 'date-picker';
  placeholder?: string;
  required?: boolean;
  disabled?: boolean;
  
  // Validation
  minLength?: number;
  maxLength?: number;
  min?: number;
  max?: number;
  step?: number;     // a number, never 'any'; without one, a number input steps by 10^-scale, or 'any' with no scale
  pattern?: string;
  
  // Options (for select, radio-group)
  options?: Array<{ value: string; label: string }>;

  // Styling: the classes of the control this field draws, joined with its own; never the field row
  className?: string;
}

interface FormSchema {
  type: 'form';
  fields: FormField[];            // Form fields
  defaultValues?: Record<string, any>;  // Initial values, keyed by field name (a field has no default key of its own)
  submitLabel?: string;           // Submit button text (default 'Submit')
  
  // Events
  // RUNTIME SLOT (objectui#6124): a React host supplies it; JSON refuses it
  // by name. Author behaviour as a node type, e.g. an action:button node.
  onSubmit?: (data: Record<string, any>) => void | Promise<void>;
  
  // Layout
  columns?: number;               // Grid layout
  
  // Styling
  className?: string;
}
```
