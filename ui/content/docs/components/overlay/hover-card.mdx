The Hover Card component displays rich content when hovering over an element.

## Basic Usage

<SchemaExample id="components-overlay-hover-card/basic-hover-card" />

## Schema

```ts
import type { SchemaNode } from '@object-ui/types';

interface HoverCardSchema {
  type: 'hover-card';
  trigger: SchemaNode | SchemaNode[];   // Trigger element
  content: SchemaNode | SchemaNode[];   // Card content
  side?: 'top' | 'right' | 'bottom' | 'left';
  align?: 'start' | 'center' | 'end';
  openDelay?: number;                   // Delay before opening (ms)
  closeDelay?: number;                  // Delay before closing (ms)
  className?: string;
}
```
