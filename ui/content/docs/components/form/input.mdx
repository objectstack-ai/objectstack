## Schema

```ts
interface InputSchema {
  type: 'input';
  name?: string;
  label?: string;
  inputType?: 'text' | 'password' | 'email' | 'number' | 'tel' | 'url';
  placeholder?: string;
  defaultValue?: string | number;
  required?: boolean;
  disabled?: boolean | string | { dialect?: string; source: string };   // boolean, predicate expression, or CEL envelope
  readonly?: boolean;
}
```
