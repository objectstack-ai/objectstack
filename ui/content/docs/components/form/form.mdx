## Schema

```ts
interface FormField {
  name: string;
  label?: string;
  type: 'input' | 'textarea' | 'select' | 'checkbox' | 'radio-group' | 'date-picker';
  placeholder?: string;
  required?: boolean;
  disabled?: boolean;
  
  // Validation
  minLength?: number;
  maxLength?: number;
  min?: number;
  max?: number;
  step?: number;     // a number, never 'any'; without one, a number input steps by 10^-scale, or 'any' with no scale
  pattern?: string;
  
  // Options (for select, radio-group)
  options?: Array<{ value: string; label: string }>;

  // Styling: the classes of the control this field draws, joined with its own; never the field row
  className?: string;
}

interface FormSchema {
  type: 'form';
  fields: FormField[];            // Form fields
  defaultValues?: Record<string, any>;  // Initial values, keyed by field name (a field has no default key of its own)
  submitLabel?: string;           // Submit button text (default 'Submit')
  
  // Events
  // RUNTIME SLOT (objectui#6124): a React host supplies it; JSON refuses it
  // by name. Author behaviour as a node type, e.g. an action:button node.
  onSubmit?: (data: Record<string, any>) => void | Promise<void>;
  
  // Layout
  columns?: number;               // Grid layout
  
  // Styling
  className?: string;
}
```
