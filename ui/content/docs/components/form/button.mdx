## Schema

```ts
interface ButtonSchema {
  type: 'button';
  label: string;                    // Button text
  variant?: 'default' | 'destructive' | 'outline' | 'secondary' | 'ghost' | 'link';
  size?: 'sm' | 'default' | 'lg' | 'icon';
  
  // Icons
  icon?: string;                    // Lucide icon name
  iconPosition?: 'left' | 'right'; // Icon placement
  
  // States
  disabled?: boolean | string | { dialect?: string; source: string };   // boolean, predicate expression, or CEL envelope
  loading?: boolean;
  
  // Actions
  // RUNTIME SLOT (objectui#6124): a React host supplies it; JSON refuses it
  // by name. Author behaviour as a node type, e.g. an action:button node.
  onClick?: () => void | Promise<void>;  // Click handler
  
  // Styling
  className?: string;               // Tailwind classes, e.g. "w-full" for a full-width button
}
```
