## Schema

```ts
import type { SchemaNode } from '@object-ui/types';

interface EmptySchema {
  type: 'empty';
  icon?: string;                  // Lucide icon name
  title?: string;                 // Main message
  description?: string;           // Supporting text
  action?: SchemaNode;            // Call to action button (any node; a bare string renders as text)
  className?: string;
}
```
